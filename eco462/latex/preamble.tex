% --- NOTEBOX WITH OPTIONAL COLOR AND UPDATED BEHAVIOR ---
% Usage: \begin{notebox}{Title} for inherited color
%        \begin{notebox}[black]{Title} for a black frame that makes the next proof/note have a black line
\newtcolorbox{notebox}[2][\g_current_tcolorbox_color_tl]{
  enhanced,
  breakable=true,
  before skip=9pt,
  after skip=9pt,
  sharp corners,
  colback=white,
  colframe=#1, % Takes color from optional argument #1
  boxrule=1pt,
  before upper={\noindent\textcolor{#1}{\textbf{#2}}\hspace{1em}}, % Title from mandatory argument #2, colored with #1
  % If the frame color is 'black', set the next proof/note color to black.
  after={\ifstrequal{#1}{black}{\gdef\g_current_tcolorbox_color_tl{black}}{}}
}
